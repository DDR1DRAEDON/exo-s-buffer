cmake_minimum_required(VERSION 3.21)
project(PredictiveBufferV5 VERSION 0.3.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(PB_BUILD_GEODE "Build Geode Windows x64 mod" OFF)
include(CTest)
if(BUILD_TESTING)
    add_executable(pb_tests tests/test_engine.cpp)
    target_include_directories(pb_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
    if(MSVC)
        target_compile_options(pb_tests PRIVATE /W4)
    else()
        target_compile_options(pb_tests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME pb_tests COMMAND pb_tests)
endif()

if(PB_BUILD_GEODE)
    if(NOT WIN32)
        message(FATAL_ERROR "This adapter targets Geometry Dash Windows 64-bit only. Use a Win64 Geode toolchain.")
    endif()
    if(NOT DEFINED ENV{GEODE_SDK} OR "$ENV{GEODE_SDK}" STREQUAL "")
        message(FATAL_ERROR "GEODE_SDK environment variable is required")
    endif()
    add_library(predictivebuffer_v5 SHARED src/geode_adapter.cpp)
    target_include_directories(predictivebuffer_v5 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
    add_subdirectory("$ENV{GEODE_SDK}" "${CMAKE_CURRENT_BINARY_DIR}/geode")
    setup_geode_mod(predictivebuffer_v5)
endif()
